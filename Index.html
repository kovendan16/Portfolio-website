// ==================== SMOOTH PARALLAX BLOBS ==================== window.addEventListener('scroll', () => { const scrollY = window.scrollY; const blobs = document.querySelectorAll('.blob'); blobs[0].style.transform = `translate(${Math.sin(scrollY * 0.002) * 30}px, ${scrollY * 0.05}px)`; blobs[1].style.transform = `translate(${Math.cos(scrollY * 0.003) * 20}px, ${scrollY * -0.03}px)`; blobs[2].style.transform = `translate(${Math.sin(scrollY * 0.001) * 25}px, ${scrollY * 0.04}px)`; }); // ==================== INIT ==================== document.addEventListener('DOMContentLoaded', () => { // Trigger initial reveal for elements in viewport setTimeout(() => { document.querySelectorAll('.fade-up, .fade-left, .fade-right, .scale-in').forEach(el => { const rect = el.getBoundingClientRect(); if (rect.top < window.innerHeight) { el.classList.add('visible'); } }); }, 700); });
